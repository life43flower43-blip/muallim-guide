    27→  LAB_EXAM_SCENARIOS,
    28→  labScenarioById,
    29→  isExamScenario,
    30→  skillReports,
    31→  skillOfConcept,
    32→  STATUS_WORD,
    33→  labStatusColor,
    34→  RATING_WORD,
    35→  type LabAttempt,
    36→  type LabRating,
    37→  type LabScenario,
    38→  type QuizHint,
    39→} from "@/lib/lab";
    40→import { conceptById, questionById, OPTION_LETTERS } from "@/lib/qbank";
    41→
    42→type LabTab = "list" | "exam" | "map" | "mirror";
    43→
    44→const ratingIcon: Record<LabRating, typeof CheckCircle2> = {
    45→  good: CheckCircle2,
    46→  ok: MinusCircle,
    47→  poor: CircleAlert,

    54→
    55→export default function LabView() {
    56→  const clearLab = useUi((s) => s.clearLab);
    57→  const startScenarios = useUi((s) => s.startScenarios);
    58→  const openConcept = useUi((s) => s.openConcept);
    59→  const labAttempts = useApp((s) => s.labAttempts);
    60→  const addLabAttempt = useApp((s) => s.addLabAttempt);

    62→
    63→  const [tab, setTab] = useState<LabTab>("list");

    65→  const [examMode, setExamMode] = useState(false);

    68→  const [steps, setSteps] = useState<LabAttempt["steps"]>([]);
    69→  const [pendingOutcome, setPendingOutcome] = useState<LabAttempt["steps"][number] | null>(null);

    90→    setPendingOutcome(null);
    91→    setReflect({ success: "", failed: "", change: "" });
    92→  };
    93→
    94→  const node = scenario ? scenario.nodes[nodeKey] : null;

   103→    for (let i = arr.length - 1; i > 0; i--) {
   104→      h = (h * 1103515245 + 12345) >>> 0;
   105→      const j = h % (i + 1);
   106→      [arr[i], arr[j]] = [arr[j], arr[i]];
   107→    }
   108→    return arr;
   109→  }, [node, scenario, runId]);

   122→    setPendingOutcome(step);
   123→  };

   125→  const proceed = () => {

   129→    if (!opt) return;
   130→    const nextSteps = [...steps, pendingOutcome];
   131→    setSteps(nextSteps);
   132→    setPendingOutcome(null);
   133→    setNodeKey(opt.next === "end" ? "end" : opt.next);
   134→  };
   135→
   136→  const saveAttempt = () => {

   139→    setScenario(null);
   140→    setTab(examMode ? "exam" : "list");
   141→  };
   142→
   143→  const openConceptDeep = (cid: string) => {

   145→    openConcept(cid);
   146→  };
   147→
   148→  const header = (t: LabTab) => (

   150→      <div className="flex items-center justify-between">
   151→        <div>
   152→          <h1 className="text-xl font-black text-foreground">مختبر المعلّم</h1>
   153→          <p className="text-xs text-muted-foreground">تدرّب على قرار المعلم في مواقف الحصة الحقيقية</p>

   155→        <button
   156→          onClick={clearLab}
   157→          aria-label="إغلاق"
   158→          className="rounded-full bg-muted p-2 text-muted-foreground hover:bg-accent"
   159→        >
   160→          <X className="h-5 w-5" />
   161→        </button>
   162→      </div>
   163→      <div className="mt-3 flex gap-1.5">
   164→        {(
   165→          [
   166→            ["list", "المواقف", Dumbbell],
   167→            ["exam", "الاستقلال", ShieldCheck],
   168→            ["map", "الخريطة", MapIcon],
   169→            ["mirror", "مرآتي", Eye],
   170→          ] as const
   171→        ).map(([id, lbl, Icon]) => (
   172→          <button
   173→            key={id}
   174→            onClick={() => setTab(id)}
   175→            className={`flex flex-1 items-center justify-center gap-1 rounded-xl border-2 px-1.5 py-2 text-[11px] font-bold ${
   176→              t === id ? "border-primary bg-accent text-primary" : "border-border bg-white text-muted-foreground"
   177→            }`}
   178→          >
   179→            <Icon className="h-4 w-4" />
   180→            {lbl}
   181→          </button>
   182→        ))}
   183→      </div>
   184→    </div>

   194→        exit={{ opacity: 0, y: 24 }}
   195→        className="fixed inset-0 z-50 overflow-y-auto bg-background"

   197→        <div className="mx-auto max-w-lg px-4 pb-28 pt-5">
   198→          <div className="mb-4 flex items-center justify-between">
   199→            <button
   200→              onClick={() => setScenario(null)}
   201→              className="flex items-center gap-1 rounded-full bg-muted px-3 py-1.5 text-xs font-bold text-muted-foreground"
   202→            >
   203→              <ArrowLeft className="h-4 w-4" /> {examMode ? "إنهاء الاختبار" : "خروج من الموقف"}
   204→            </button>

   207→                <ShieldCheck className="h-3.5 w-3.5" /> اختبار استقلال

   210→              <span className="rounded-full bg-accent px-3 py-1 text-xs font-bold text-primary">{scenario.skill}</span>

   214→          <h2 className="text-lg font-black leading-8">{scenario.title}</h2>

   217→          {!finished && !pendingOutcome && node && (
   218→            <motion.div key={nodeKey + runId} initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} className="mt-4 space-y-3">
   219→              <div className="rounded-2xl border-2 border-primary/20 bg-white p-4 text-sm font-bold leading-7">{node.text}</div>

   228→                    {OPTION_LETTERS[i]}
   229→                  </span>
   230→                  <span className="text-sm font-bold leading-6">{o.text}</span>
   231→                </button>
   232→              ))}
   233→            </motion.div>
   234→          )}
   235→
   236→          {pendingOutcome && (
   237→            <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} className="mt-4 space-y-3">
   238→              <div className={`flex items-center gap-2 rounded-2xl border-2 p-3 text-sm font-black ${ratingCls[pendingOutcome.rating]}`}>
   239→                {(() => {
   240→                  const Icon = ratingIcon[pendingOutcome.rating];
   241→                  return <Icon className="h-5 w-5" />;
   242→                })()}
   243→                {RATING_WORD[pendingOutcome.rating]}
   244→              </div>
   245→              <div className="rounded-2xl border border-border bg-white p-4 text-sm leading-7">{pendingOutcome.outcome}</div>

   247→                onClick={proceed}

   254→
   255→          {finished && (

   259→                  <ShieldCheck className="h-5 w-5" /> نتيجة اختبار الاستقلال — قراراتك بلا تلميح الآن بلا تقييم
   260→                </div>
   261→              )}
   262→              <div className="rounded-2xl border-2 border-primary/30 bg-accent p-4">
   263→                <p className="mb-3 flex items-center gap-2 text-sm font-black text-primary">
   264→                  <Sparkles className="h-4 w-4" /> قراراتك في هذا الموقف
   265→                </p>
   266→                <div className="space-y-2">
   267→                  {steps.map((s, i) => (
   268→                    <div key={i} className="rounded-xl bg-white p-3 text-xs leading-6">
   269→                      <span className={`mr-1 font-black ${s.rating === "good" ? "text-indigo-700" : s.rating === "ok" ? "text-amber-700" : "text-red-700"}`}>
   270→                        {RATING_WORD[s.rating]}:
   271→                      </span>
   272→                      <span className="font-bold">{s.choice}</span>
   273→                    </div>
   274→                  ))}
   275→                </div>
   276→              </div>
   277→              <div className="rounded-2xl border border-border bg-white p-4 text-sm leading-7">
   278→                <p className="mb-2 flex items-center gap-1.5 text-sm font-black text-primary">
   279→                  <Quote className="h-4 w-4" /> خلاصة الموقف
   280→                </p>
   281→                {scenario.end}
   282→              </div>
   283→
   284→              <div>
   285→                <p className="mb-2 flex items-center gap-1.5 text-sm font-black">
   286→                  <BookOpen className="h-4 w-4 text-primary" /> مفاهيم من المكتبة تخدم هذا الموقف
   287→                </p>
   288→                <div className="flex flex-wrap gap-2">
   289→                  {scenario.conceptIds.map((cid) => {
   290→                    const c = conceptById(cid);
   291→                    if (!c) return null;
   292→                    return (
   293→                      <button
   294→                        key={cid}
   295→                        onClick={() => openConceptDeep(cid)}
   296→                        className="rounded-full border border-primary/30 bg-white px-3 py-1.5 text-xs font-bold text-primary"
   297→                      >
   298→                        {c.title}
   299→                      </button>
   300→                    );
   301→                  })}
   302→                </div>
   303→              </div>
   304→
   305→              <div className="rounded-2xl border-2 border-border bg-white p-4">
   306→                <p className="mb-3 flex items-center gap-2 text-sm font-black">
   307→                  <Eye className="h-4 w-4 text-primary" /> مرآة المعلّم — تأمل بعد التجربة
   308→                </p>
   309→                <div className="space-y-3">
   310→                  {(
   311→                    [
   312→                      ["success", "ما الذي نجح في تصرفاتك؟"],
   313→                      ["failed", "ما القرار الذي لم ينجح؟ ولماذا؟"],
   314→                      ["change", "ما الذي سأغيّره في المرة المقبلة؟"],
   315→                    ] as const
   316→                  ).map(([k, lbl]) => (
   317→                    <div key={k}>
   318→                      <label className="mb-1 block text-xs font-bold text-muted-foreground">{lbl}</label>
   319→                      <textarea
   320→                        value={reflect[k]}
   321→                        onChange={(e) => setReflect((r) => ({ ...r, [k]: e.target.value }))}
   322→                        rows={2}
   323→                        className="w-full rounded-xl border-2 border-border bg-background p-3 text-sm leading-6 outline-none focus:border-primary"
   324→                      />
   325→                    </div>
   326→                  ))}
   327→                </div>
   328→                <button
   329→                  onClick={saveAttempt}
   330→                  className="mt-4 flex w-full items-center justify-center gap-2 rounded-2xl bg-primary py-3 font-black text-primary-foreground"
   331→                >
   332→                  <CheckCircle2 className="h-5 w-5" /> {examMode ? "احفظ نتيجة الاختبار والتأمل" : "احفظ التجربة والتأمل"}
   333→                </button>
   334→              </div>
   335→            </motion.div>
   336→          )}
   337→        </div>
   338→      </motion.div>
   339→    );
   340→  }
   341→
   342→  // ---------- تبويب المواقف التدريبية ----------
   343→  if (tab === "list") {
   344→    return (
   345→      <motion.div
   346→        initial={{ opacity: 0, y: 24 }}
   347→        animate={{ opacity: 1, y: 0 }}
   348→        exit={{ opacity: 0, y: 24 }}
   349→        className="fixed inset-0 z-50 overflow-y-auto bg-background"
   350→      >
   351→        {header(tab)}
   352→        <div className="mx-auto max-w-lg px-4 pb-28 pt-4">
   353→          <div className="mb-4 rounded-2xl border-2 border-primary/20 bg-gradient-to-l from-accent to-white p-4">
   354→            <p className="flex items-center gap-2 text-sm font-black text-primary">
   355→              <FlaskConical className="h-4 w-4" /> كيف يعمل المختبر؟
   356→            </p>
   357→            <p className="mt-1 text-xs leading-6 text-muted-foreground">
   358→              تدخل موقفاً صفياً حقيقياً، وتتخذ قراراتك خطوة بخطوة، فيتحول الموقف بأثرها، ثم ترى تحليل كل قرار وتتأمل في أدائك في «مرآة المعلّم».

   360→          </div>
   361→
   362→          <button
   363→            onClick={() => {
   364→              clearLab();

   367→            className="mb-4 flex w-full items-center justify-between rounded-2xl border-2 border-border bg-white p-4 text-right"

   370→              <p className="text-sm font-black">مواقف سريعة</p>

   376→          <div className="space-y-3">

   404→    return (

   406→        initial={{ opacity: 0, y: 24 }}
   407→        animate={{ opacity: 1, y: 0 }}
   408→        exit={{ opacity: 0, y: 24 }}

   418→              هنا مواقف جديدة لم تتدرّب عليها ولا تقرأ حلولها. أثناء القرار لا يظهر تقييم ولا نتيجة ولا شرح — التحليل الكامل يفتح عند الختام وحده. ما تحققه هنا هو دليلك الحقيقي على الاستقلال المهني، وهو ما تثبته في خريطة الكفاءات.

   426→                  key={s.id}
   427→                  onClick={() => startScenario(s, true)}

   587→// أدلة نوعية من سجل الأسئلة: لكل كفاءة، مفاهيمها المرتبطة عبر المواقف وآخر إجابة عن أسئلة كل مفهوم
   588→function quizHintsOf(quizAttempts: Record<string, { correct: boolean; ts: number }>): QuizHint[] {
   589→  const bySkill = new Map<string, { wrong: string[]; good: string[] }>();
   590→  for (const [, skills] of skillOfConcept) for (const sk of skills) if (!bySkill.has(sk)) bySkill.set(sk, { wrong: [], good: [] });
   591→  for (const [qid, a] of Object.entries(quizAttempts)) {
   592→    const q = questionById(qid);
   593→    if (!q) continue;
   594→    const skills = skillOfConcept.get(q.conceptId);
   595→    if (!skills?.length) continue;
   596→    const title = conceptById(q.conceptId)?.title;
   597→    if (!title) continue;
   598→    for (const sk of skills) {
   599→      const b = bySkill.get(sk);
   600→      if (!b) continue;
   601→      const arr = a.correct ? b.good : b.wrong;
   602→      if (!arr.includes(title)) arr.push(title);
   603→    }
   604→  }
   605→  return [...bySkill.entries()].map(([skill, b]) => ({ skill, wrong: b.wrong, good: b.good }));
   606→}
