    40→function ScoreRing({ pct, size = 150 }: { pct: number; size?: number }) {
    41→  const r = (size - 18) / 2;
    42→  const c = 2 * Math.PI * r;
    43→  const color = pct >= 80 ? "#2D6A4F" : pct >= 50 ? "#F59E0B" : "#E5484D";
    44→  return (
    45→    <div className="relative" style={{ width: size, height: size }}>
    46→      <svg width={size} height={size} className="-rotate-90">
    47→        <circle cx={size / 2} cy={size / 2} r={r} stroke="#E7ECE6" strokeWidth="14" fill="none" />
    48→        <motion.circle
    49→          cx={size / 2}
    50→          cy={size / 2}
    51→          r={r}
    52→          stroke={color}
    53→          strokeWidth="14"
    54→          fill="none"
    55→          strokeLinecap="round"
    56→          strokeDasharray={c}
    57→          initial={{ strokeDashoffset: c }}
    58→          animate={{ strokeDashoffset: c * (1 - pct / 100) }}
    59→          transition={{ duration: 1, ease: "easeOut" }}
    60→        />
    61→      </svg>
    62→      <div className="absolute inset-0 flex flex-col items-center justify-center">
    63→        <span className="text-3xl font-black" style={{ color }}>
    64→          {ad(Math.round(pct))}٪
    65→        </span>
    66→      </div>
    67→    </div>
    68→  );
    69→}

    71→function Timer({ seconds, onTimeout }: { seconds: number; onTimeout: () => void }) {
    72→  const [left, setLeft] = useState(seconds);
    73→  const firedRef = useRef(false);
    74→  useEffect(() => {
    75→    const t = setInterval(() => {
    76→      setLeft((v) => {
    77→        if (v <= 1) {
    78→          clearInterval(t);
    79→          if (!firedRef.current) {
    80→            firedRef.current = true;
    81→            onTimeout();
    82→          }
    83→          return 0;
    84→        }
    85→        return v - 1;
    86→      });
    87→    }, 1000);
    88→    return () => clearInterval(t);
    89→  }, []);
    90→  const m = Math.floor(left / 60);
    91→  const s = left % 60;
    92→  const danger = left <= 60;
    93→  return (
    94→    <div
    95→      className={`flex items-center gap-1 rounded-full px-3 py-1 text-sm font-bold tabular-nums ${
    96→        danger ? "bg-red-100 text-red-700 pulse-ring" : "bg-secondary text-primary"
    97→      }`}
    98→    >
    99→      <TimerIcon className="h-4 w-4" />
   100→      <span dir="ltr">
   101→        {ad(m)}:{ad(String(s).padStart(2, "0"))}
   102→      </span>
   103→    </div>
   104→  );
   105→}

   107→function SocraticBox({ q }: { q: Question }) {
   108→  const [stepIdx, setStepIdx] = useState(0);
   109→  const [showHint, setShowHint] = useState(false);
   110→  const steps: SocraticStep[] = useMemo(() => buildSocraticSteps(q), [q]);
   111→  const done = stepIdx >= steps.length;
   112→
   113→  return (
   114→    <div className="rounded-2xl border border-amber-200 bg-amber-50 p-4">
   115→      <div className="mb-3 flex items-center gap-2">
   116→        <div className="flex h-8 w-8 items-center justify-center rounded-full bg-amber-400">
   117→          <Lightbulb className="h-4 w-4 text-amber-950" />
   118→        </div>
   119→        <p className="text-sm font-extrabold text-amber-900">المدرّب السقراطي — فكّر معي</p>
   120→      </div>
   121→      {!done ? (
   122→        <div className="space-y-3">
   123→          <p className="font-bold text-foreground">{steps[stepIdx].prompt}</p>
   124→          {!showHint ? (
   125→            <Button
   126→              size="sm"
   127→              variant="outline"
   128→              className="rounded-full border-amber-300 bg-white text-amber-900"
   129→              onClick={() => setShowHint(true)}
   130→            >
   131→              <Brain className="ml-1 h-4 w-4" /> جرب أول نفسي
   132→            </Button>
   133→          ) : (
   134→            <motion.div initial={{ opacity: 0, y: 6 }} animate={{ opacity: 1, y: 0 }}>
   135→              <p className="rounded-xl border border-amber-200 bg-white p-3 text-sm leading-7 text-amber-950">
   136→                {steps[stepIdx].hint}
   137→              </p>
   138→            </motion.div>
   139→          )}
   140→          {showHint && (
   141→            <Button
   142→              size="sm"
   143→              className="rounded-full bg-amber-500 text-white hover:bg-amber-600"
   144→              onClick={() => {
   145→                setStepIdx((i) => i + 1);
   146→                setShowHint(false);
   147→              }}
   148→            >
   149→              {stepIdx + 1 < steps.length ? "سؤال تالٍ" : "فتح الإجابة الكاملة"}
   150→              <ArrowLeft className="mr-1 h-4 w-4" />
   151→            </Button>
   152→          )}
   153→        </div>
   154→      ) : (
   155→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="space-y-2">
   156→          <p className="text-sm leading-7 text-foreground">
   157→            <span className="font-extrabold text-primary">الإجابة الصحيحة: </span>
   158→            {q.correct}
   159→          </p>
   160→          <p className="text-sm leading-7 text-muted-foreground">{q.reason}</p>
   161→        </motion.div>
   162→      )}
   163→    </div>
   164→  );
   165→}

   236→export default function QuizSession({ config }: { config: QuizConfig }) {
   237→  const clearQuiz = useUi((s) => s.clearQuiz);
   238→  const recordAttempt = useApp((s) => s.recordAttempt);
   239→  const finishSession = useApp((s) => s.finishSession);
   240→  const attempts = useApp((s) => s.attempts);
   241→  const reportedQuestions = useApp((s) => s.reportedQuestions);
   242→  const reportQuestion = useApp((s) => s.reportQuestion);
   243→
   244→  const [idx, setIdx] = useState(0);
   245→  const [selected, setSelected] = useState<string | null>(null);
   246→  const [graded, setGraded] = useState<null | boolean>(null);
   247→  const [socratic, setSocratic] = useState(false);
   248→  const [exiting, setExiting] = useState(false);
   249→  const [finished, setFinished] = useState(false);
   250→  const [recorded, setRecorded] = useState(false);
   251→  const [wrongShake, setWrongShake] = useState(false);
   252→  const [sessionXp, setSessionXp] = useState(0);
   253→  const [review, setReview] = useState<{ qid: string; ok: boolean }[]>([]);
   254→  const startedAt = useRef(Date.now());
   255→
   256→  const questions = useMemo(
   257→    () => config.qids.map((id) => questionById(id)).filter(Boolean) as Question[],
   258→    [config.qids]
   259→  );
   260→  const q = questions[idx];
   261→  const instantFeedback = config.mode !== "exam";
   262→  const concept = q ? conceptById(q.conceptId) : undefined;
   263→
   264→  // عند انتهاء الجلسة: تسجيل النقاط مرة واحدة
   265→  useEffect(() => {
   266→    if (finished && !recorded) {
   267→      setRecorded(true);
   268→      const correctCount = review.filter((r) => r.ok).length;
   269→      const durationSec = Math.round((Date.now() - startedAt.current) / 1000);
   270→      finishSession({
   271→        mode: config.mode,
   272→        label: config.label,
   273→        total: questions.length,
   274→        correctCount,
   275→        xp: sessionXp,
   276→        durationSec,
   277→      });
   278→    }
   279→  }, [finished, recorded, review, sessionXp, config, questions.length, finishSession]);
   280→
   281→  if (questions.length === 0) {
   282→    return (
   283→      <div className="flex h-dvh flex-col items-center justify-center gap-4 bg-background p-8 text-center">
   284→        <Brain className="h-14 w-14 text-primary" />
   285→        <h2 className="text-xl font-black">لا توجد أسئلة في هذه الجلسة</h2>
   286→        <p className="text-sm text-muted-foreground">
   287→          {config.mode === "review"
   288→            ? "لا توجد أسئلة مستحقة للمراجعة الآن — أنجز تدريباً جديداً وستظهر المراجعة المتباعدة لاحقاً."
   289→            : "جرّب نمطاً آخر من شاشة الاختبار."}
   290→        </p>
   291→        <Button onClick={clearQuiz} className="rounded-full bg-primary px-8 text-primary-foreground">
   292→          رجوع
   293→        </Button>
   294→      </div>
   295→    );
   296→  }
   297→
   298→  const grade = () => {
   299→    if (!q || selected === null) return;
   300→    const ok = selected === q.correct;
   301→    setGraded(ok);
   302→    if (!ok) setWrongShake(true);
   303→    setTimeout(() => setWrongShake(false), 450);
   304→    recordAttempt(q.id, ok, selected ?? undefined);
   305→    const wasUnseen = !attempts[q.id];
   306→    const gain = ok
   307→      ? wasUnseen
   308→        ? q.difficulty === "صعب"
   309→          ? 15
   310→          : q.difficulty === "متوسط"
   311→            ? 10
   312→            : 8
   313→        : 6
   314→      : 0;
   315→    setSessionXp((x) => x + gain);
   316→    setReview((r) => [...r, { qid: q.id, ok }]);
   317→  };
   318→
   319→  const next = () => {
   320→    setSelected(null);
   321→    setGraded(null);
   322→    setSocratic(false);
   323→    if (idx + 1 >= questions.length) {
   324→      const correctCount = review.filter((r) => r.ok).length;
   325→      if (config.mode === "exam" && correctCount === questions.length) {
   326→        setSessionXp((x) => x + 30);
   327→      } else if (correctCount === questions.length && questions.length >= 5) {
   328→        setSessionXp((x) => x + 20);
   329→      }
   330→      setFinished(true);
   331→    } else {
   332→      setIdx((i) => i + 1);
   333→    }
   334→  };
   335→
   336→  const exit = () => {
   337→    if (finished || review.length === 0) {
   338→      clearQuiz();
   339→    } else {
   340→      setExiting(true);
   341→    }
   342→  };
   343→
   344→  // زر «أبلغ عن خطأ» — يُحفظ معرف السؤال محليًا ويختفي الزر للسؤال المبلّغ عنه
   345→  const reported = q ? reportedQuestions.includes(q.id) : false;
   346→  const handleReport = () => {
   347→    if (!q || reported) return;
   348→    reportQuestion(q.id);
   349→    toast({ title: "شكرًا — سيُراجع السؤال" });
   350→  };
   351→
   352→  // ---------- شاشة النتائج ----------
   353→  if (finished) {
   354→    const correctCount = review.filter((r) => r.ok).length;
   355→    const pct = (correctCount / questions.length) * 100;
   356→    const msg =
   357→      pct === 100
   358→        ? "إتقان كامل! أنت أسطورة"
   359→        : pct >= 80
   360→          ? "أداء ممتاز، استمر!"
   361→          : pct >= 50
   362→            ? "جيد — راجع أخطائك وستتحسن"
   363→            : "لا بأس، التوجيه بعد الخطأ طريق الإتقان";
   364→    return (
   365→      <div className="flex h-dvh flex-col bg-background">
   366→        <div className="safe-top flex items-center justify-between px-4 pb-2">
   367→          <p className="text-lg font-black text-primary">النتيجة</p>
   368→          <button onClick={clearQuiz} className="rounded-full bg-muted p-2" aria-label="إغلاق">
   369→            <X className="h-5 w-5" />
   370→          </button>
   371→        </div>
   372→        <div className="no-scrollbar flex-1 overflow-y-auto px-4 pb-8">
   373→          <div className="pop-in flex flex-col items-center gap-3 py-4">
   374→            <ScoreRing pct={pct} />
   375→            <h2 className="text-xl font-black">{msg}</h2>
   376→            <div className="flex flex-wrap justify-center gap-3">
   377→              <div className="flex items-center gap-2 rounded-full bg-secondary px-4 py-2 text-sm font-bold text-primary">
   378→                <CheckCircle2 className="h-4 w-4" /> {ad(correctCount)} من {ad(questions.length)} إجابة صحيحة
   379→              </div>
   380→            </div>
   381→            {review.length < questions.length && (
   382→              <p className="rounded-2xl bg-red-50 px-4 py-2 text-xs font-bold leading-6 text-red-700">
   383→                لم تُجب عن {arCount(questions.length - review.length, QUESTION_OBJ_FORMS)} قبل انتهاء
   384→                الوقت — احتُسبت خطأً كالامتحان الحقيقي.
   385→              </p>
   386→            )}
   387→          </div>
   388→          <p className="mb-2 mt-4 px-1 text-sm font-extrabold text-muted-foreground">
   389→            مراجعة الأسئلة — اضغط لعرض التفسير
   390→          </p>
   391→          <div className="space-y-2">
   392→            {review.map((r) => {
   393→              const rq = questionById(r.qid)!;
   394→              return (
   395→                <details key={r.qid} className="card-soft rounded-2xl border border-border bg-white p-4">
   396→                  <summary className="flex cursor-pointer list-none items-start gap-3">
   397→                    {r.ok ? (
   398→                      <CheckCircle2 className="mt-0.5 h-5 w-5 shrink-0 text-emerald-600" />
   399→                    ) : (
   400→                      <XCircle className="mt-0.5 h-5 w-5 shrink-0 text-red-500" />
   401→                    )}
   402→                    <span className="text-sm font-bold leading-6">{rq.question}</span>
   403→                  </summary>
   404→                  {!r.ok && (
   405→                    <p className="mt-2 rounded-lg bg-emerald-50 px-3 py-2 text-xs font-bold text-emerald-800">
   406→                      الإجابة الصحيحة: {rq.correct}
   407→                    </p>
   408→                  )}
   409→                  <p className="mt-2 text-xs leading-6 text-muted-foreground">{rq.reason}</p>
   410→                </details>
   411→              );
   412→            })}
   413→          </div>
   414→        </div>
   415→        <div className="safe-bottom border-t border-border bg-white px-4 pt-3">
   416→          <Button
   417→            onClick={clearQuiz}
   418→            className="h-14 w-full rounded-2xl bg-primary text-lg font-black text-primary-foreground"
   419→          >
   420→            <Trophy className="ml-2 h-5 w-5" /> متابعة رحلة التعلم
   421→          </Button>
   422→        </div>
   423→      </div>
   424→    );
   425→  }
   426→
   427→  // ---------- شاشة السؤال ----------
   428→  const progress = (idx / questions.length) * 100;
   429→
   430→  return (
   431→    <div className="tap-none flex h-dvh flex-col bg-background">
   432→      {/* رأس الجلسة */}
   433→      <div className="safe-top px-4 pb-2 pt-2">
   434→        <div className="mb-2 flex items-center gap-3">
   435→          <button onClick={exit} className="rounded-full p-1.5 text-muted-foreground" aria-label="خروج">
   436→            <X className="h-6 w-6" />
   437→          </button>
   438→          <div className="progress-track h-3.5 flex-1">
   439→            <div className="progress-fill h-full" style={{ width: `${progress}%` }} />
   440→          </div>
   441→          {config.timeLimitSec ? (
   442→            <Timer seconds={config.timeLimitSec} onTimeout={() => setFinished(true)} />
   443→          ) : (
   444→            <span className="rounded-full bg-secondary px-3 py-1 text-sm font-bold tabular-nums text-primary">
   445→              {ad(idx + 1)} من {ad(questions.length)}
   446→            </span>
   447→          )}
   448→        </div>
   449→        <p className="text-center text-xs font-bold text-muted-foreground">{config.label}</p>
   450→      </div>
   451→
   452→      {/* جسم السؤال */}
   453→      <div className="no-scrollbar flex-1 overflow-y-auto px-4 pb-4">
   454→        <AnimatePresence mode="wait">
   455→          <motion.div
   456→            key={q.id}
   457→            initial={{ opacity: 0, x: -30 }}
   458→            animate={{ opacity: 1, x: 0 }}
   459→            exit={{ opacity: 0, x: 30 }}
   460→            transition={{ duration: 0.25 }}
   461→            className="space-y-4"
   462→          >
   463→            <div className="flex flex-wrap items-center gap-2 pt-1">
   464→              <span className="rounded-full bg-accent px-3 py-1 text-xs font-bold text-primary">
   465→                {q.type === "mcq" ? "اختيار من متعدد" : "صح أم خطأ"}
   466→              </span>
   467→              <span
   468→                className={`rounded-full px-3 py-1 text-xs font-bold ${
   469→                  q.difficulty === "سهل"
   470→                    ? "bg-emerald-100 text-emerald-700"
   471→                    : q.difficulty === "متوسط"
   472→                      ? "bg-amber-100 text-amber-700"
   473→                      : "bg-red-100 text-red-600"
   474→                }`}
   475→              >
   476→                {q.difficulty}
   477→              </span>
   478→              {concept && (
   479→                <span className="rounded-full bg-muted px-3 py-1 text-xs font-medium text-muted-foreground">
   480→                  {concept.title}
   481→                </span>
   482→              )}
   483→            </div>
   484→
   485→            <h2 className="text-xl font-extrabold leading-9">{q.question}</h2>
   486→
   487→            {q.type === "tf" ? (
   488→              <div className="grid grid-cols-2 gap-3">
   489→                {[q.options[0], q.options[1]].map((o, i) => {
