   115→// قسم نصي بعرض الخطة — أسطر مفصولة تُعرض شرطات، ويختفي القسم إن كان فارغًا
   116→function PlanLines({ title, body }: { title: string; body: string }) {
   117→  const lines = body.split(/\n+/).map((s) => s.trim()).filter(Boolean);
   118→  if (!lines.length) return null;
   119→  return (
   120→    <div className="rounded-2xl border-2 border-border bg-white p-4">
   121→      <h3 className="text-sm font-black">{title}</h3>
   122→      <div className="mt-2 space-y-1">
   123→        {lines.map((ln, i) => (
   124→          <p key={i} className="text-xs font-bold leading-6">
   125→            – {ln}
   126→          </p>
   127→        ))}
   128→      </div>
   129→    </div>
   130→  );
   131→}
