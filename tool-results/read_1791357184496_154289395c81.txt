    40→    40→function ScoreRing({ pct, size = 150 }: { pct: number; size?: number }) {
    41→    41→  const r = (size - 18) / 2;
    42→    42→  const c = 2 * Math.PI * r;
    43→    43→  const color = pct >= 80 ? "#2D6A4F" : pct >= 50 ? "#F59E0B" : "#E5484D";
    44→    44→  return (
    45→    45→    <div className="relative" style={{ width: size, height: size }}>
    46→    46→      <svg width={size} height={size} className="-rotate-90">
    47→    47→        <circle cx={size / 2} cy={size / 2} r={r} stroke="#E7ECE6" strokeWidth="14" fill="none" />
    48→    48→        <motion.circle
    49→    49→          cx={size / 2}
    50→    50→          cy={size / 2}
    51→    51→          r={r}
    52→    52→          stroke={color}
    53→    53→          strokeWidth="14"
    54→    54→          fill="none"
    55→    55→          strokeLinecap="round"
    56→    56→          strokeDasharray={c}
    57→    57→          initial={{ strokeDashoffset: c }}
    58→    58→          animate={{ strokeDashoffset: c * (1 - pct / 100) }}
    59→    59→          transition={{ duration: 1, ease: "easeOut" }}
    60→    60→        />
    61→    61→      </svg>
    62→    62→      <div className="absolute inset-0 flex flex-col items-center justify-center">

    64→    64→          {ad(Math.round(pct))}٪

    67→    67→    </div>

    71→    71→function Timer({ seconds, onTimeout }: { seconds: number; onTimeout: () => void }) {
    72→    72→  const [left, setLeft] = useState(seconds);
    73→    73→  const firedRef = useRef(false);
    74→    74→  useEffect(() => {
    75→    75→    const t = setInterval(() => {
    76→    76→      setLeft((v) => {
    77→    77→        if (v <= 1) {
    78→    78→          clearInterval(t);
    79→    79→          if (!firedRef.current) {
    80→    80→            firedRef.current = true;
    81→    81→            onTimeout();
    82→    82→          }
    83→    83→          return 0;
    84→    84→        }
    85→    85→        return v - 1;
    86→    86→      });
    87→    87→    }, 1000);
    88→    88→    return () => clearInterval(t);
    89→    89→  }, []);
    90→    90→  const m = Math.floor(left / 60);
    91→    91→  const s = left % 60;
    92→    92→  const danger = left <= 60;
    93→    93→  return (
    94→    94→    <div
    95→    95→      className={`flex items-center gap-1 rounded-full px-3 py-1 text-sm font-bold tabular-nums ${
    96→    96→        danger ? "bg-red-100 text-red-700 pulse-ring" : "bg-secondary text-primary"
    97→    97→      }`}
    98→    98→    >
    99→    99→      <TimerIcon className="h-4 w-4" />
   100→   100→      <span dir="ltr">
   101→   101→        {ad(m)}:{ad(String(s).padStart(2, "0"))}
   102→   102→      </span>
   103→   103→    </div>
   104→   104→  );
   105→   105→}

   242→   242→  const reportQuestion = useApp((s) => s.reportQuestion);
   243→   243→
   244→   244→  const [idx, setIdx] = useState(0);
   245→   245→  const [selected, setSelected] = useState<string | null>(null);
   246→   246→  const [graded, setGraded] = useState<null | boolean>(null);
   247→   247→  const [socratic, setSocratic] = useState(false);
   248→   248→  const [exiting, setExiting] = useState(false);
   249→   249→  const [finished, setFinished] = useState(false);
   250→   250→  const [recorded, setRecorded] = useState(false);
   251→   251→  const [wrongShake, setWrongShake] = useState(false);
   252→   252→  const [sessionXp, setSessionXp] = useState(0);
   253→   253→  const [review, setReview] = useState<{ qid: string; ok: boolean }[]>([]);
   254→   254→  const startedAt = useRef(Date.now());
   255→   255→
   256→   256→  const questions = useMemo(
   257→   257→    () => config.qids.map((id) => questionById(id)).filter(Boolean) as Question[],
   258→   258→    [config.qids]
   259→   259→  );
   260→   260→  const q = questions[idx];
   261→   261→  const instantFeedback = config.mode !== "exam";
   262→   262→  const concept = q ? conceptById(q.conceptId) : undefined;
   263→   263→

   265→   265→  useEffect(() => {

   267→   267→      setRecorded(true);
   268→   268→      const correctCount = review.filter((r) => r.ok).length;
   269→   269→      const durationSec = Math.round((Date.now() - startedAt.current) / 1000);
   270→   270→      finishSession({
   271→   271→        mode: config.mode,
   272→   272→        label: config.label,
   273→   273→        total: questions.length,
   274→   274→        correctCount,
   275→   275→        xp: sessionXp,
   276→   276→        durationSec,
   277→   277→      });
   278→   278→    }
   279→   279→  }, [finished, recorded, review, sessionXp, config, questions.length, finishSession]);
   280→   280→
   281→   281→  if (questions.length === 0) {
   282→   282→    return (
   283→   283→      <div className="flex h-dvh flex-col items-center justify-center gap-4 bg-background p-8 text-center">
   284→   284→        <Brain className="h-14 w-14 text-primary" />
   285→   285→        <h2 className="text-xl font-black">لا توجد أسئلة في هذه الجلسة</h2>
   286→   286→        <p className="text-sm text-muted-foreground">
   287→   287→          {config.mode === "review"
   288→   288→            ? "لا توجد أسئلة مستحقة للمراجعة الآن — أنجز تدريباً جديداً وستظهر المراجعة المتباعدة لاحقاً."

   296→   296→  }
   297→   297→

   306→   306→    const gain = ok
