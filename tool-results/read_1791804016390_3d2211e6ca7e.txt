    43→function ResultBadge({ tone }: { tone: "great" | "good" | "fair" | "weak" }) {
    44→  const toneCls =
    45→    tone === "great"
    46→      ? "bg-emerald-100 text-emerald-700"
    47→      : tone === "good"
    48→        ? "bg-emerald-100 text-emerald-700"
    49→        : tone === "fair"
    50→          ? "bg-amber-100 text-amber-700"
    51→          : "bg-red-100 text-red-600";
    52→  const Icon = tone === "weak" ? XCircle : Trophy;
    53→  return (
    54→    <span className={`flex h-24 w-24 items-center justify-center rounded-full ${toneCls}`}>
    55→      <Icon className="h-12 w-12" />
    56→    </span>
    57→  );
    58→}

    60→function Timer({ seconds, onTimeout }: { seconds: number; onTimeout: () => void }) {
    61→  const [left, setLeft] = useState(seconds);
    62→  const firedRef = useRef(false);
    63→  useEffect(() => {
    64→    const t = setInterval(() => {
    65→      setLeft((v) => {
    66→        if (v <= 1) {
    67→          clearInterval(t);
    68→          if (!firedRef.current) {
    69→            firedRef.current = true;
    70→            onTimeout();
    71→          }
    72→          return 0;
    73→        }
    74→        return v - 1;
    75→      });
    76→    }, 1000);
    77→    return () => clearInterval(t);
    78→  }, []);
    79→  const m = Math.floor(left / 60);
    80→  const s = left % 60;
    81→  const danger = left <= 60;
    82→  return (
    83→    <div
    84→      className={`flex items-center gap-1 rounded-full px-3 py-1 text-sm font-bold tabular-nums ${
    85→        danger ? "bg-red-100 text-red-700 pulse-ring" : "bg-secondary text-primary"
    86→      }`}
    87→    >
    88→      <TimerIcon className="h-4 w-4" />
    89→      <span dir="ltr">
    90→        {ad(m)}:{ad(String(s).padStart(2, "0"))}
    91→      </span>
    92→    </div>
    93→  );
    94→}

    96→function SocraticBox({ q }: { q: Question }) {
    97→  const [stepIdx, setStepIdx] = useState(0);
    98→  const [showHint, setShowHint] = useState(false);
    99→  const steps: SocraticStep[] = useMemo(() => buildSocraticSteps(q), [q]);
   100→  const done = stepIdx >= steps.length;
   101→
   102→  return (
   103→    <div className="rounded-2xl border border-amber-200 bg-amber-50 p-4">
   104→      <div className="mb-3 flex items-center gap-2">
   105→        <div className="flex h-8 w-8 items-center justify-center rounded-full bg-amber-400">
   106→          <Lightbulb className="h-4 w-4 text-amber-950" />
   107→        </div>
   108→        <p className="text-sm font-extrabold text-amber-900">المدرّب السقراطي — فكّر معي</p>
   109→      </div>
   110→      {!done ? (
   111→        <div className="space-y-3">
   112→          <p className="font-bold text-foreground">{steps[stepIdx].prompt}</p>
   113→          {!showHint ? (
   114→            <Button
   115→              size="sm"
   116→              variant="outline"
   117→              className="rounded-full border-amber-300 bg-white text-amber-900"
   118→              onClick={() => setShowHint(true)}
   119→            >
   120→              <Brain className="ml-1 h-4 w-4" /> جرب أول نفسي
   121→            </Button>
   122→          ) : (
   123→            <motion.div initial={{ opacity: 0, y: 6 }} animate={{ opacity: 1, y: 0 }}>
   124→              <p className="rounded-xl border border-amber-200 bg-white p-3 text-sm leading-7 text-amber-950">
   125→                {steps[stepIdx].hint}
   126→              </p>
   127→            </motion.div>
   128→          )}
   129→          {showHint && (
   130→            <Button
   131→              size="sm"
   132→              className="rounded-full bg-amber-500 text-white hover:bg-amber-600"
   133→              onClick={() => {
   134→                setStepIdx((i) => i + 1);
   135→                setShowHint(false);
   136→              }}
   137→            >
   138→              {stepIdx + 1 < steps.length ? "سؤال تالٍ" : "فتح الإجابة الكاملة"}
   139→              <ArrowLeft className="mr-1 h-4 w-4" />
   140→            </Button>
   141→          )}
   142→        </div>
   143→      ) : (
   144→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="space-y-2">
   145→          <p className="text-sm leading-7 text-foreground">
   146→            <span className="font-extrabold text-primary">الإجابة الصحيحة: </span>
   147→            {q.correct}
   148→          </p>
   149→          <p className="text-sm leading-7 text-muted-foreground">{q.reason}</p>
   150→        </motion.div>
   151→      )}
   152→    </div>
   153→  );
   154→}

   225→export default function QuizSession({ config }: { config: QuizConfig }) {
   226→  const clearQuiz = useUi((s) => s.clearQuiz);
   227→  const recordAttempt = useApp((s) => s.recordAttempt);
   228→  const finishSession = useApp((s) => s.finishSession);
   229→  const attempts = useApp((s) => s.attempts);
   230→  const reportedQuestions = useApp((s) => s.reportedQuestions);
   231→  const reportQuestion = useApp((s) => s.reportQuestion);
   232→
   233→  // استئناف جلسة مقطوعة — يُفعّل فقط عند الضغط على «أكمل جلستك» في مركز الاختبارات
   234→  const resumed = useMemo(() => {
   235→    if (typeof window === "undefined") return null;
   236→    try {
   237→      if (sessionStorage.getItem(RESUME_FLAG) !== "1") return null;
   238→      sessionStorage.removeItem(RESUME_FLAG);
   239→    } catch {
   240→      return null;
   241→    }
   242→    const snap = loadSnapshot();
   243→    if (!snap) return null;
   244→    if (snap.config.label !== config.label || snap.config.qids.length !== config.qids.length)
   245→      return null;
   246→    return snap;
   247→  }, [config]);
   248→
   249→  const [idx, setIdx] = useState(resumed?.idx ?? 0);
   250→  const [selected, setSelected] = useState<string | null>(null);
   251→  const [graded, setGraded] = useState<null | boolean>(null);
   252→  const [socratic, setSocratic] = useState(false);
   253→  const [exiting, setExiting] = useState(false);
   254→  const [finished, setFinished] = useState(false);
   255→  const [recorded, setRecorded] = useState(false);
   256→  const [wrongShake, setWrongShake] = useState(false);
   257→  const [sessionXp, setSessionXp] = useState(resumed?.sessionXp ?? 0);
   258→  const [review, setReview] = useState<{ qid: string; ok: boolean }[]>(resumed?.review ?? []);
   259→  const startedAt = useRef(Date.now());
   260→
   261→  const questions = useMemo(
   262→    () => config.qids.map((id) => questionById(id)).filter(Boolean) as Question[],
   263→    [config.qids]
   264→  );
   265→  const q = questions[idx];
   266→  const instantFeedback = config.mode !== "exam";
   267→  const concept = q ? conceptById(q.conceptId) : undefined;
   268→  // خلط ترتيب خيارات المقالي بثبات لكل سؤال — لا يُخلط صح/خطأ (أيقوناته دلالية)
   269→  const mcqOpts = useMemo(() => (q && q.type === "mcq" ? shuffle(q.options) : []), [q]);
   270→
   271→  // عند انتهاء الجلسة: تسجيل النقاط مرة واحدة
   272→  useEffect(() => {
   273→    if (finished && !recorded) {
   274→      setRecorded(true);
   275→      const correctCount = review.filter((r) => r.ok).length;
   276→      const durationSec = Math.round((Date.now() - startedAt.current) / 1000);
   277→      finishSession({
   278→        mode: config.mode,
   279→        label: config.label,
   280→        total: questions.length,
   281→        correctCount,
   282→        xp: sessionXp,
   283→        durationSec,
   284→      });
   285→      clearSnapshot(); // انتهت الجلسة — لا لقطة للاستئناف
   286→    }
   287→  }, [finished, recorded, review, sessionXp, config, questions.length, finishSession]);
   288→
   289→  // حفظ لقطة الجلسة بعد كل سؤال — تُستأنف من مركز الاختبارات عند العودة
   290→  useEffect(() => {
   291→    if (finished || questions.length === 0) return;
   292→    saveSnapshot({ config, idx, review, sessionXp });
   293→  }, [finished, config, idx, review, sessionXp, questions.length]);
   294→
   295→  if (questions.length === 0) {
   296→    return (
   297→      <div className="flex h-dvh flex-col items-center justify-center gap-4 bg-background p-8 text-center">
   298→        <Brain className="h-14 w-14 text-primary" />
   299→        <h2 className="text-xl font-black">لا توجد أسئلة في هذه الجلسة</h2>
   300→        <p className="text-sm text-muted-foreground">
   301→          {config.mode === "review"
   302→            ? "لا توجد أسئلة مستحقة للمراجعة الآن — أنجز تدريباً جديداً وستظهر المراجعة المتباعدة لاحقاً."
   303→            : "جرّب نمطاً آخر من شاشة الاختبار."}
   304→        </p>
   305→        <Button onClick={clearQuiz} className="rounded-full bg-primary px-8 text-primary-foreground">
   306→          رجوع
   307→        </Button>
   308→      </div>
   309→    );
   310→  }
   311→
   312→  const grade = () => {
   313→    if (!q || selected === null) return;
   314→    const ok = selected === q.correct;
   315→    setGraded(ok);
   316→    // اهتزاز لطيف يؤكد القرار لمسيًا — نجاح قصير واحد، خطأ نبضتان متقطعتان (ويحترم مفتاح الإعدادات)
   317→    try {
   318→      if ("vibrate" in navigator && useApp.getState().haptics)
   319→        navigator.vibrate(ok ? 35 : [60, 40, 60]);
   320→    } catch {}
   321→    if (!ok) setWrongShake(true);
   322→    setTimeout(() => setWrongShake(false), 450);
   323→    recordAttempt(q.id, ok, selected ?? undefined);
   324→    const wasUnseen = !attempts[q.id];
   325→    const gain = ok
   326→      ? wasUnseen
   327→        ? q.difficulty === "صعب"
   328→          ? 15
   329→          : q.difficulty === "متوسط"
   330→            ? 10
   331→            : 8
   332→        : 6
   333→      : 0;
   334→    setSessionXp((x) => x + gain);
   335→    setReview((r) => [...r, { qid: q.id, ok }]);
   336→  };
   337→
   338→  const next = () => {
   339→    setSelected(null);
   340→    setGraded(null);
   341→    setSocratic(false);
   342→    if (idx + 1 >= questions.length) {
   343→      const correctCount = review.filter((r) => r.ok).length;
   344→      if (config.mode === "exam" && correctCount === questions.length) {
   345→        setSessionXp((x) => x + 30);
   346→      } else if (correctCount === questions.length && questions.length >= 5) {
   347→        setSessionXp((x) => x + 20);
   348→      }
   349→      setFinished(true);
   350→    } else {
   351→      setIdx((i) => i + 1);
   352→    }
   353→  };
   354→
   355→  const exit = () => {
   356→    if (finished || review.length === 0) {
   357→      clearSnapshot(); // خروج صريح بلا تقدم ذي به — لا استئناف
   358→      clearQuiz();
   359→    } else {
   360→      setExiting(true);
   361→    }
   362→  };
   363→
   364→  // زر «أبلغ عن خطأ» — يُحفظ معرف السؤال محليًا ويختفي الزر للسؤال المبلّغ عنه
   365→  const reported = q ? reportedQuestions.includes(q.id) : false;
   366→  const handleReport = () => {
   367→    if (!q || reported) return;
   368→    reportQuestion(q.id);
   369→    toast({ title: "شكرًا — سيُراجع السؤال" });
   370→  };
   371→
   372→  // ---------- شاشة النتائج ----------
   373→  if (finished) {
   374→    const correctCount = review.filter((r) => r.ok).length;
   375→    const pct = (correctCount / questions.length) * 100;
   376→    const msg =
   377→      pct === 100
   378→        ? "إتقان كامل! أنت أسطورة"
   379→        : pct >= 80
   380→          ? "أداء ممتاز، استمر!"
   381→          : pct >= 50
   382→            ? "جيد — راجع أخطائك وستتحسن"
   383→            : "لا بأس، التوجيه بعد الخطأ طريق الإتقان";
   384→    const tone: "great" | "good" | "fair" | "weak" =
   385→      pct === 100 ? "great" : pct >= 80 ? "good" : pct >= 50 ? "fair" : "weak";
   386→    return (
   387→      <div className="flex h-dvh flex-col bg-background">
   388→        <div className="safe-top flex items-center justify-between px-4 pb-2">
   389→          <p className="text-lg font-black text-primary">النتيجة</p>
   390→          <button onClick={clearQuiz} className="rounded-full bg-muted p-2" aria-label="إغلاق">
   391→            <X className="h-5 w-5" />
   392→          </button>
   393→        </div>
   394→        <div className="no-scrollbar flex-1 overflow-y-auto px-4 pb-8">
   395→          <div className="pop-in flex flex-col items-center gap-3 py-4">
   396→            <ResultBadge tone={tone} />
   397→            <h2 className="text-xl font-black">{msg}</h2>
   398→            {review.length < questions.length && (
   399→              <p className="rounded-2xl bg-red-50 px-4 py-2 text-xs font-bold leading-6 text-red-700">
   400→                لم تُجب عن بعض الأسئلة قبل انتهاء الوقت — احتُسبت خطأً كالامتحان الحقيقي.
   401→              </p>
   402→            )}
   403→          </div>
   404→          <p className="mb-2 mt-4 px-1 text-sm font-extrabold text-muted-foreground">
   405→            مراجعة الأسئلة — اضغط لعرض التفسير
   406→          </p>
   407→          <div className="space-y-2">
   408→            {review.map((r) => {
   409→              const rq = questionById(r.qid)!;
   410→              return (
   411→                <details key={r.qid} className="card-soft rounded-2xl border border-border bg-white p-4">
   412→                  <summary className="flex cursor-pointer list-none items-start gap-3">
   413→                    {r.ok ? (
   414→                      <CheckCircle2 className="mt-0.5 h-5 w-5 shrink-0 text-emerald-600" />
   415→                    ) : (
   416→                      <XCircle className="mt-0.5 h-5 w-5 shrink-0 text-red-500" />
   417→                    )}
   418→                    <span className="text-sm font-bold leading-6">{rq.question}</span>
   419→                  </summary>
   420→                  {!r.ok && (
   421→                    <p className="mt-2 rounded-lg bg-emerald-50 px-3 py-2 text-xs font-bold text-emerald-800">
   422→                      الإجابة الصحيحة: {rq.correct}
   423→                    </p>
   424→                  )}
   425→                  <p className="mt-2 text-xs leading-6 text-muted-foreground">{rq.reason}</p>
   426→                </details>
   427→              );
   428→            })}
   429→          </div>
   430→        </div>
   431→        <div className="safe-bottom border-t border-border bg-white px-4 pt-3">
   432→          <Button
   433→            onClick={clearQuiz}
   434→            className="h-14 w-full rounded-2xl bg-primary text-lg font-black text-primary-foreground"
   435→          >
   436→            <Trophy className="ml-2 h-5 w-5" /> متابعة رحلة التعلم
   437→          </Button>
   438→        </div>
   439→      </div>
   440→    );
   441→  }
   442→
   443→  // ---------- شاشة السؤال ----------
   444→  const progress = (idx / questions.length) * 100;
   445→
   446→  return (
   447→    <div className="tap-none flex h-dvh flex-col bg-background">
   448→      {/* رأس الجلسة */}
   449→      <div className="safe-top px-4 pb-2 pt-2">
   450→        <div className="mb-2 flex items-center gap-3">
   451→          <button onClick={exit} className="rounded-full p-1.5 text-muted-foreground" aria-label="خروج">
   452→            <X className="h-6 w-6" />
   453→          </button>
   454→          <div className="progress-track h-3.5 flex-1">
   455→            <div className="progress-fill h-full" style={{ width: `${progress}%` }} />
   456→          </div>
   457→          {config.timeLimitSec ? (
   458→            <Timer seconds={config.timeLimitSec} onTimeout={() => setFinished(true)} />
   459→          ) : (
   460→            <span className="h-8" aria-hidden />
   461→          )}
   462→        </div>
   463→        <p className="text-center text-xs font-bold text-muted-foreground">{config.label}</p>
   464→      </div>
   465→
   466→      {/* جسم السؤال */}
   467→      <div className="no-scrollbar flex-1 overflow-y-auto px-4 pb-4">
   468→        <AnimatePresence mode="wait">
   469→          <motion.div
   470→            key={q.id}
   471→            initial={{ opacity: 0, x: -30 }}
   472→            animate={{ opacity: 1, x: 0 }}
   473→            exit={{ opacity: 0, x: 30 }}
   474→            transition={{ duration: 0.25 }}
   475→            className="space-y-4"
   476→          >
