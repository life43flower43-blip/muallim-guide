    60→    60→function Timer({ seconds, onTimeout }: { seconds: number; onTimeout: () => void }) {
    61→    61→  const [left, setLeft] = useState(seconds);
    62→    62→  const firedRef = useRef(false);
    63→    63→  useEffect(() => {
    64→    64→    const t = setInterval(() => {
    65→    65→      setLeft((v) => {
    66→    66→        if (v <= 1) {
    67→    67→          clearInterval(t);
    68→    68→          if (!firedRef.current) {
    69→    69→            firedRef.current = true;
    70→    70→            onTimeout();
    71→    71→          }
    72→    72→          return 0;
    73→    73→        }
    74→    74→        return v - 1;
    75→    75→      });
    76→    76→    }, 1000);
    77→    77→    return () => clearInterval(t);
    78→    78→  }, []);
    79→    79→  const m = Math.floor(left / 60);
    80→    80→  const s = left % 60;
    81→    81→  const danger = left <= 60;
    82→    82→  return (
    83→    83→    <div
    84→    84→      className={`flex items-center gap-1 rounded-full px-3 py-1 text-sm font-bold tabular-nums ${
    85→    85→        danger ? "bg-red-100 text-red-700 pulse-ring" : "bg-secondary text-primary"
    86→    86→      }`}
    87→    87→    >
    88→    88→      <TimerIcon className="h-4 w-4" />
    89→    89→      <span dir="ltr">
    90→    90→        {ad(m)}:{ad(String(s).padStart(2, "0"))}
    91→    91→      </span>
    92→    92→    </div>
    93→    93→  );
    94→    94→}

   225→   225→export default function QuizSession({ config }: { config: QuizConfig }) {
   226→   226→  const clearQuiz = useUi((s) => s.clearQuiz);
   227→   227→  const recordAttempt = useApp((s) => s.recordAttempt);
   228→   228→  const finishSession = useApp((s) => s.finishSession);
   229→   229→  const attempts = useApp((s) => s.attempts);
   230→   230→  const reportedQuestions = useApp((s) => s.reportedQuestions);
   231→   231→  const reportQuestion = useApp((s) => s.reportQuestion);
   232→   232→
   233→   233→  // استئناف جلسة مقطوعة — يُفعّل فقط عند الضغط على «أكمل جلستك» في مركز الاختبارات
   234→   234→  const resumed = useMemo(() => {
   235→   235→    if (typeof window === "undefined") return null;
   236→   236→    try {
   237→   237→      if (sessionStorage.getItem(RESUME_FLAG) !== "1") return null;
   238→   238→      sessionStorage.removeItem(RESUME_FLAG);
   239→   239→    } catch {
   240→   240→      return null;
   241→   241→    }
   242→   242→    const snap = loadSnapshot();
   243→   243→    if (!snap) return null;
   244→   244→    if (snap.config.label !== config.label || snap.config.qids.length !== config.qids.length)
   245→   245→      return null;
   246→   246→    return snap;
   247→   247→  }, [config]);
   248→   248→
   249→   249→  const [idx, setIdx] = useState(resumed?.idx ?? 0);
   250→   250→  const [selected, setSelected] = useState<string | null>(null);
   251→   251→  const [graded, setGraded] = useState<null | boolean>(null);
   252→   252→  const [socratic, setSocratic] = useState(false);
   253→   253→  const [exiting, setExiting] = useState(false);
   254→   254→  const [finished, setFinished] = useState(false);
   255→   255→  const [recorded, setRecorded] = useState(false);
   256→   256→  const [wrongShake, setWrongShake] = useState(false);
   257→   257→  const [sessionXp, setSessionXp] = useState(resumed?.sessionXp ?? 0);
   258→   258→  const [review, setReview] = useState<{ qid: string; ok: boolean }[]>(resumed?.review ?? []);
   259→   259→  const startedAt = useRef(Date.now());
   260→   260→
   261→   261→  const questions = useMemo(
   262→   262→    () => config.qids.map((id) => questionById(id)).filter(Boolean) as Question[],
   263→   263→    [config.qids]
   264→   264→  );
   265→   265→  const q = questions[idx];
   266→   266→  const instantFeedback = config.mode !== "exam";
   267→   267→  const concept = q ? conceptById(q.conceptId) : undefined;
   268→   268→  // خلط ترتيب خيارات المقالي بثبات لكل سؤال — لا يُخلط صح/خطأ (أيقوناته دلالية)
   269→   269→  const mcqOpts = useMemo(() => (q && q.type === "mcq" ? shuffle(q.options) : []), [q]);
   270→   270→
   271→   271→  // عند انتهاء الجلسة: تسجيل النقاط مرة واحدة
   272→   272→  useEffect(() => {
   273→   273→    if (finished && !recorded) {
   274→   274→      setRecorded(true);
   275→   275→      const correctCount = review.filter((r) => r.ok).length;
   276→   276→      const durationSec = Math.round((Date.now() - startedAt.current) / 1000);
   277→   277→      finishSession({
   278→   278→        mode: config.mode,
   279→   279→        label: config.label,
   280→   280→        total: questions.length,
   281→   281→        correctCount,
   282→   282→        xp: sessionXp,
   283→   283→        durationSec,
   284→   284→      });
   285→   285→      clearSnapshot(); // انتهت الجلسة — لا لقطة للاستئناف
   286→   286→    }
   287→   287→  }, [finished, recorded, review, sessionXp, config, questions.length, finishSession]);
   288→   288→
   289→   289→  // حفظ لقطة الجلسة بعد كل سؤال — تُستأنف من مركز الاختبارات عند العودة
   290→   290→  useEffect(() => {
   291→   291→    if (finished || questions.length === 0) return;
   292→   292→    saveSnapshot({ config, idx, review, sessionXp });
   293→   293→  }, [finished, config, idx, review, sessionXp, questions.length]);
   294→   294→
   295→   295→  if (questions.length === 0) {
   296→   296→    return (
   297→   297→      <div className="flex h-dvh flex-col items-center justify-center gap-4 bg-background p-8 text-center">
   298→   298→        <Brain className="h-14 w-14 text-primary" />
   299→   299→        <h2 className="text-xl font-black">لا توجد أسئلة في هذه الجلسة</h2>
   300→   300→        <p className="text-sm text-muted-foreground">
   301→   301→          {config.mode === "review"
   302→   302→            ? "لا توجد أسئلة مستحقة للمراجعة الآن — أنجز تدريباً جديداً وستظهر المراجعة المتباعدة لاحقاً."
   303→   303→            : "جرّب نمطاً آخر من شاشة الاختبار."}
   304→   304→        </p>
   305→   305→        <Button onClick={clearQuiz} className="rounded-full bg-primary px-8 text-primary-foreground">
   306→   306→          رجوع
   307→   307→        </Button>
   308→   308→      </div>
   309→   309→    );
   310→   310→  }
   311→   311→
   312→   312→  const grade = () => {
   313→   313→    if (!q || selected === null) return;
   314→   314→    const ok = selected === q.correct;
   315→   315→    setGraded(ok);
   316→   316→    // اهتزاز لطيف يؤكد القرار لمسيًا — نجاح قصير واحد، خطأ نبضتان متقطعتان (ويحترم مفتاح الإعدادات)
   317→   317→    try {
   318→   318→      if ("vibrate" in navigator && useApp.getState().haptics)
   319→   319→        navigator.vibrate(ok ? 35 : [60, 40, 60]);
   320→   320→    } catch {}
   321→   321→    if (!ok) setWrongShake(true);
   322→   322→    setTimeout(() => setWrongShake(false), 450);
   323→   323→    recordAttempt(q.id, ok, selected ?? undefined);
   324→   324→    const wasUnseen = !attempts[q.id];
   325→   325→    const gain = ok
   326→   326→      ? wasUnseen
   327→   327→        ? q.difficulty === "صعب"
   328→   328→          ? 15
   329→   329→          : q.difficulty === "متوسط"
   330→   330→            ? 10
   331→   331→            : 8
   332→   332→        : 6
   333→   333→      : 0;
   334→   334→    setSessionXp((x) => x + gain);
   335→   335→    setReview((r) => [...r, { qid: q.id, ok }]);
   336→   336→  };
   337→   337→
   338→   338→  const next = () => {
   339→   339→    setSelected(null);
   340→   340→    setGraded(null);
   341→   341→    setSocratic(false);
   342→   342→    if (idx + 1 >= questions.length) {
   343→   343→      const correctCount = review.filter((r) => r.ok).length;
   344→   344→      if (config.mode === "exam" && correctCount === questions.length) {
   345→   345→        setSessionXp((x) => x + 30);
   346→   346→      } else if (correctCount === questions.length && questions.length >= 5) {
   347→   347→        setSessionXp((x) => x + 20);
   348→   348→      }
   349→   349→      setFinished(true);
   350→   350→    } else {
   351→   351→      setIdx((i) => i + 1);
   352→   352→    }
   353→   353→  };
   354→   354→
   355→   355→  const exit = () => {
   356→   356→    if (finished || review.length === 0) {
   357→   357→      clearSnapshot(); // خروج صريح بلا تقدم ذي به — لا استئناف
   358→   358→      clearQuiz();
   359→   359→    } else {
   360→   360→      setExiting(true);
   361→   361→    }
   362→   362→  };
   363→   363→
   364→   364→  // زر «أبلغ عن خطأ» — يُحفظ معرف السؤال محليًا ويختفي الزر للسؤال المبلّغ عنه
   365→   365→  const reported = q ? reportedQuestions.includes(q.id) : false;
   366→   366→  const handleReport = () => {
   367→   367→    if (!q || reported) return;
   368→   368→    reportQuestion(q.id);
   369→   369→    toast({ title: "شكرًا — سيُراجع السؤال" });
   370→   370→  };
   371→   371→
   372→   372→  // ---------- شاشة النتائج ----------
   373→   373→  if (finished) {
   374→   374→    const correctCount = review.filter((r) => r.ok).length;
   375→   375→    const pct = (correctCount / questions.length) * 100;
   376→   376→    const msg =
   377→   377→      pct === 100
   378→   378→        ? "إتقان كامل! أنت أسطورة"
   379→   379→        : pct >= 80
   380→   380→          ? "أداء ممتاز، استمر!"
   381→   381→          : pct >= 50
   382→   382→            ? "جيد — راجع أخطائك وستتحسن"
   383→   383→            : "لا بأس، التوجيه بعد الخطأ طريق الإتقان";
   384→   384→    const tone: "great" | "good" | "fair" | "weak" =
   385→   385→      pct === 100 ? "great" : pct >= 80 ? "good" : pct >= 50 ? "fair" : "weak";
   386→   386→    return (
   387→   387→      <div className="flex h-dvh flex-col bg-background">
   388→   388→        <div className="safe-top flex items-center justify-between px-4 pb-2">
   389→   389→          <p className="text-lg font-black text-primary">النتيجة</p>
   390→   390→          <button onClick={clearQuiz} className="rounded-full bg-muted p-2" aria-label="إغلاق">
   391→   391→            <X className="h-5 w-5" />
   392→   392→          </button>
   393→   393→        </div>
   394→   394→        <div className="no-scrollbar flex-1 overflow-y-auto px-4 pb-8">
   395→   395→          <div className="pop-in flex flex-col items-center gap-3 py-4">
   396→   396→            <ResultBadge tone={tone} />
   397→   397→            <h2 className="text-xl font-black">{msg}</h2>
   398→   398→            {review.length < questions.length && (
   399→   399→              <p className="rounded-2xl bg-red-50 px-4 py-2 text-xs font-bold leading-6 text-red-700">
   400→   400→                لم تُجب عن بعض الأسئلة قبل انتهاء الوقت — احتُسبت خطأً كالامتحان الحقيقي.
   401→   401→              </p>
   402→   402→            )}
