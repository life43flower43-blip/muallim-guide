   535→// بناء جولة: غير المحلول أولاً ثم عشوائي، بحد أقصى n
   536→export const buildScenarioRun = (solvedIds: string[], n = 10): Scenario[] => {
   537→  const solved = new Set(solvedIds);
   538→  const fresh = SCENARIOS.filter((s) => !solved.has(s.id));
   539→  const done = SCENARIOS.filter((s) => solved.has(s.id));
   540→  return [...fresh, ...done].slice(0, Math.min(n, SCENARIOS.length));
   541→};
