    47→/** تقسيم نص عربي طويل إلى فقرات قراءة مريحة على حدود الجُمل (٢–٣ جمل لكل فقرة) */
    48→function splitArParagraphs(text: string): string[] {
    49→  const t = (text ?? "").trim();
    50→  if (!t) return [];
    51→  const sentences = t.split(/(?<=[.؟!…])\s+/).filter(Boolean);
    52→  if (sentences.length < 3 || t.length < 220) return [t];
    53→  const paras: string[] = [];
    54→  let cur = "";
    55→  for (const s of sentences) {
    56→    if (cur && cur.length > 130) {
    57→      paras.push(cur.trim());
    58→      cur = s;
    59→    } else {
    60→      cur = cur ? `${cur} ${s}` : s;
    61→    }
    62→  }
    63→  if (cur.trim()) paras.push(cur.trim());
    64→  return paras.length ? paras : [t];
    65→}

   262→  onClose: () => void;

   269→  for (const c of COURSES) {

   271→    if (found) {
   272→      concept = { concept: found, course: c };

   304→            <X className="h-5 w-5" />
   305→          </button>
   306→        </div>
   307→      </div>

   336→                <SpeakButton text={`${c.title}. ${c.explanation}`} label="استمع للشرح" />
   337→              </div>

   404→      </div>

   654→            hint={COURSE_SHORT[openCourse.id]}
   655→            accent={COURSE_COLOR[openCourse.id] ?? "#5B21B6"}

   694→      {/* التفاصيل تُعرض فوق أي مستوى */}
